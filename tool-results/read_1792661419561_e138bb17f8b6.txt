   484→   484→        setVisionBusy(false)

   560→   560→    a.click()
   561→   561→    URL.revokeObjectURL(url)
   562→   562→    toast({ title: 'خروجی گرفتی! 📥', description: 'فایل چت دانلود شد.' })

   564→   564→

   613→   613→              height: s.size,

   615→   615→              animationDelay: s.delay,

   617→   617→          />
   618→   618→        ))}
   619→   619→      </div>
   620→   620→
   621→   621→      <TekaSidebar
   622→   622→        conversations={conversations}
   623→   623→        activeId={activeId}
   624→   624→        loading={loadingList}
