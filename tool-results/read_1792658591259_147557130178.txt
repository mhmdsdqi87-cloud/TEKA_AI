   518→        title: 'اووپس!',
   519→        description: err instanceof Error ? err.message : 'یه خطا پیش اومد!',

   562→    toast({ title: 'خروجی گرفتی! 📥', description: 'فایل چت دانلود شد.' })
   563→  }, [messages, toast])
   564→
   565→  // ---------- تغییر تم ----------
   566→  const toggleTheme = useCallback(() => {
   567→    setTheme(theme === 'dark' ? 'light' : 'dark')
   568→  }, [theme, setTheme])
   569→
   570→  // آخرین جواب تکا (برای دکمه دوباره بگو)
   571→  const lastAssistantId = useMemo(() => {
   572→    for (let i = messages.length - 1; i >= 0; i--) {
   573→      if (messages[i].role === 'assistant') return messages[i].id
   574→    }
   575→    return null
   576→  }, [messages])
   577→
   578→  const busy = sending || regenerating || imageBusy || visionBusy
   579→
   580→  // ستاره‌های پس‌زمینه (ثابت برای جلوگیری از خطای hydration)
   581→  const stars = useMemo(
   582→    () => [
   583→      { top: '12%', left: '8%', size: 3, dur: '3.6s', delay: '0s' },
   584→      { top: '22%', left: '82%', size: 2, dur: '4.4s', delay: '0.6s' },
   585→      { top: '38%', left: '15%', size: 2, dur: '5s', delay: '1.2s' },
   586→      { top: '55%', left: '88%', size: 3, dur: '4s', delay: '0.3s' },
   587→      { top: '68%', left: '6%', size: 2, dur: '4.8s', delay: '1.8s' },
   588→      { top: '78%', left: '70%', size: 2, dur: '3.8s', delay: '0.9s' },
   589→      { top: '30%', left: '45%', size: 2, dur: '5.2s', delay: '2.1s' },
   590→      { top: '85%', left: '35%', size: 3, dur: '4.2s', delay: '1.5s' },
   591→      { top: '8%', left: '55%', size: 2, dur: '4.6s', delay: '2.4s' },
   592→      { top: '48%', left: '60%', size: 2, dur: '3.9s', delay: '0.4s' },
   593→    ],
   594→    []
   595→  )
   596→
   597→  return (
   598→    <div className="relative flex h-dvh w-full overflow-hidden">
   599→      {/* پس‌زمینهٔ aurora + ستاره‌ها */}
   600→      <div className="teka-aurora" aria-hidden="true">
   601→        <div className="teka-blob teka-blob-1" />
   602→        <div className="teka-blob teka-blob-2" />
   603→        <div className="teka-blob teka-blob-3" />
   604→        <div className="teka-blob teka-blob-4" />
   605→        {stars.map((s, i) => (
   606→          <span
   607→            key={i}
   608→            className="teka-star"
   609→            style={{
   610→              top: s.top,
   611→              left: s.left,
   612→              width: s.size,
   613→              height: s.size,
   614→              ['--dur' as string]: s.dur,
   615→              animationDelay: s.delay,
   616→            }}
   617→          />
   618→        ))}
   619→      </div>
   620→
   621→      <TekaSidebar
   622→        conversations={conversations}
   623→        activeId={activeId}
   624→        loading={loadingList}
   625→        open={sidebarOpen}
   626→        onClose={() => setSidebarOpen(false)}
   627→        onSelect={openConversation}
   628→        onNewChat={newChat}
   629→        onDelete={deleteConversation}
   630→      />
   631→
   632→      <div className="relative z-10 flex min-w-0 flex-1 flex-col">
   633→        {/* هدر */}
   634→        <header className="flex items-center gap-2 border-b border-border/60 bg-background/70 px-3 py-2.5 backdrop-blur-xl md:px-6">
   635→          <Button
   636→            variant="ghost"
   637→            size="icon"
   638→            className="md:hidden"
   639→            onClick={() => setSidebarOpen(true)}
   640→            aria-label="باز کردن تاریخچه چت‌ها"
   641→          >
   642→            <Menu className="size-5" />
   643→          </Button>
   644→          <div className="flex min-w-0 flex-1 items-center gap-2.5">
   645→            <span className="relative flex size-2.5 shrink-0" aria-hidden="true">
   646→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-primary opacity-60" />
   647→              <span className="relative inline-flex size-2.5 rounded-full bg-primary" />
   648→            </span>
   649→            <div className="min-w-0">
   650→              <p className="truncate text-sm font-bold leading-5">تکا</p>
   651→              <p className="truncate text-[11px] leading-4 text-muted-foreground">
   652→                همیشه آنلاین • حافظه داره
   653→              </p>
   654→            </div>
   655→          </div>
   656→
   657→          {/* حالت قدرتمند */}
   658→          <Button
   659→            variant={power ? 'default' : 'ghost'}
   660→            size="sm"
   661→            onClick={() => setPower((p) => !p)}
   662→            className={`gap-1.5 rounded-full text-xs font-semibold transition-all ${
   663→              power ? 'shadow-lg shadow-primary/30' : 'text-muted-foreground'
   664→            }`}
   665→            aria-pressed={power}
   666→            title="حالت قدرتمند: جواب‌های عمیق‌تر و باهوش‌تر"
   667→          >
   668→            <Zap className={`size-4 ${power ? 'fill-current' : ''}`} />
   669→            <span className="hidden sm:inline">{power ? 'قدرتمند' : 'عادی'}</span>
   670→          </Button>
   671→
   672→          {/* خواندن خودکار */}
   673→          <Button
   674→            variant={autoSpeak ? 'default' : 'ghost'}
   675→            size="icon"
   676→            onClick={() => {
   677→              setAutoSpeak((v) => {
   678→                if (v) stopSpeaking()
   679→                return !v
   680→              })
   681→            }}
   682→            className={autoSpeak ? 'shadow-lg shadow-primary/30' : 'text-muted-foreground'}
   683→            aria-pressed={autoSpeak}
   684→            aria-label="خواندن خودکار جواب‌ها"
   685→            title="خودکار بلند بخون جواب‌هارو"
   686→          >
   687→            {mounted && autoSpeak ? <Volume2 className="size-4.5" /> : <VolumeX className="size-4.5" />}
   688→          </Button>
   689→
   690→          {/* خروجی چت */}
   691→          <Button
   692→            variant="ghost"
   693→            size="icon"
   694→            onClick={exportChat}
   695→            className="text-muted-foreground"
   696→            aria-label="دانلود خروجی چت"
   697→            title="خروجی چت"
   698→          >
   699→            <Download className="size-4.5" />
   700→          </Button>
   701→
   702→          {/* تم */}
   703→          <Button
   704→            variant="ghost"
   705→            size="icon"
   706→            onClick={toggleTheme}
   707→            className="text-muted-foreground"
   708→            aria-label={mounted && theme === 'dark' ? 'تم روشن' : 'تم تیره'}
   709→            title="تغییر تم"
   710→          >
   711→            {mounted && theme === 'dark' ? <Sun className="size-4.5" /> : <Moon className="size-4.5" />}
   712→          </Button>
   713→
   714→          <span className="hidden rounded-full border border-primary/30 bg-primary/10 px-3 py-1 text-[11px] font-medium text-primary lg:inline">
   715→            ساختهٔ محمد صادقی
   716→          </span>
   717→        </header>
   718→
   719→        {/* پیام‌ها */}
   720→        <main
   721→          ref={mainRef}
   722→          onScroll={handleMainScroll}
   723→          className="teka-scroll relative flex-1 overflow-y-auto"
   724→          aria-label="گفتگو"
   725→        >
   726→          <div className="mx-auto max-w-3xl px-3 py-4 md:px-6">
   727→            {messages.length === 0 && !loadingMessages && !busy ? (
   728→              <EmptyState onSuggestion={send} power={power} />
   729→            ) : (
   730→              <div className="flex flex-col gap-4 pb-2">
   731→                {loadingMessages &&
   732→                  Array.from({ length: 3 }).map((_, i) => (
   733→                    <div
   734→                      key={i}
   735→                      className={`flex ${i % 2 ? 'justify-end' : 'justify-start'}`}
   736→                      aria-hidden="true"
   737→                    >
   738→                      <div className="h-10 w-2/3 animate-pulse rounded-2xl bg-muted/50" />
   739→                    </div>
   740→                  ))}
   741→                {!loadingMessages &&
   742→                  messages.map((m) => (
   743→                    <MessageBubble
   744→                      key={m.id}
   745→                      message={m}
   746→                      isLastAssistant={m.id === lastAssistantId && !busy}
   747→                      onCopy={copyText}
   748→                      onSpeak={speak}
   749→                      onRegenerate={regenerate}
   750→                      speakingId={speakingId}
   751→                      speakingLoadingId={speakingLoadingId}
   752→                    />
   753→                  ))}
   754→                {busy && <TypingIndicator power={power} painting={imageBusy} analyzing={visionBusy} />}
   755→              </div>
   756→            )}
   757→            <div ref={bottomRef} />
   758→          </div>
   759→
   760→          {/* دکمه اسکرول به پایین */}
   761→          <AnimatePresence>
   762→            {showScrollBtn && (
   763→              <motion.div
   764→                initial={{ opacity: 0, y: 10 }}
   765→                animate={{ opacity: 1, y: 0 }}
   766→                exit={{ opacity: 0, y: 10 }}
   767→                className="sticky bottom-4 flex justify-center"
   768→              >
   769→                <Button
   770→                  size="icon"
   771→                  onClick={() => bottomRef.current?.scrollIntoView({ behavior: 'smooth' })}
   772→                  className="size-10 rounded-full border bg-card shadow-xl"
   773→                  aria-label="برو پایین"
   774→                >
   775→                  <ArrowDown className="size-4" />
   776→                </Button>
   777→              </motion.div>
   778→            )}
   779→          </AnimatePresence>
   780→        </main>
   781→
   782→        {/* اینپوت (فوتر چسبیده به پایین) */}
   783→        <ChatInput
   784→          value={input}
   785→          onChange={setInput}
   786→          onSend={send}
   787→          onSendImage={sendImage}
   788→          onSendVision={sendVision}
   789→          onSendFile={sendFile}
   790→          sending={busy}
   791→          analyzing={visionBusy}
   792→          power={power}
   793→        />
   794→      </div>
   795→    </div>
   796→  )
   797→}
