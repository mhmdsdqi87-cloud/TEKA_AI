   553→            (m: ChatMessage) => m.role === 'assistant'

   557→          /* شبکه یه لحظه قاطی کرد — ادامه بده */
   558→        }
   559→      }
   560→      return []
   561→    },
   562→    []
   563→  )

   565→  // ---------- ارسال پیام (⚡ استریم زنده — حرف‌به‌حرف) ----------

   714→                convId = (d.conversationId as string) ?? convId
   715→                title = (d.title as string) ?? title

   718→              }
   719→            } catch {
   720→              /* تکهٔ ناقص SSE — رد شو */
   721→            }
   722→          }
   723→        }
   724→
   725→        if (gotError) throw new Error(gotError)
   726→

   750→        }
   751→      } catch (err) {
   752→        setMessages((m) =>
   753→          m.filter((x) => x.id !== tempId && x.id !== streamMsgId)
   754→        )
   755→        setInput(text)
   756→        toast({
   757→          title: 'اوه اوه! 🙈',
   758→          description: err instanceof Error ? err.message : 'یه خطای ناشناخته پیش اومد!',
   759→          variant: 'destructive',
   760→        })
   761→      } finally {
   762→        setSending(false)
   763→        setStreamingId(null)
   764→      }
   765→    },
   766→    [sending, regenerating, imageBusy, visionBusy, power, bumpConversation, toast, stopSpeaking, waitForNewMessages, trackEvent]
