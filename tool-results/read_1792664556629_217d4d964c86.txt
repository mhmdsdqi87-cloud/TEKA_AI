   307→            `/api/conversations/${convId}?after=${encodeURIComponent(sinceIso)}`,
   308→            { cache: 'no-store', signal: AbortSignal.timeout(20_000) }

   313→            (m: ChatMessage) => m.role === 'assistant'

   549→        })
   550→        const data = await res.json().catch(() => null)
   551→        if (!res.ok) {
   552→          throw new Error(data?.error || 'آنالیز فایل نشد!')
   553→        }
   554→
   555→        if (activeIdRef.current === usedConvId) {
   556→          setMessages((m) => [...m, data.reply])
   557→          if (autoSpeakRef.current) {
   558→            speakRef.current?.(data.reply)
   559→          }
   560→        }
   561→        setActiveId(data.conversationId)
   562→        bumpConversation(data.conversationId, data.title)
   563→      } catch (err) {
   564→        setMessages((m) => m.filter((x) => x.id !== tempId))
   565→        if (question) setInput(question)
   566→        toast({
   567→          title: 'اوه اوه! 🙈',
   568→          description: err instanceof Error ? err.message : 'آنالیز فایل قاطی شد!',
   569→          variant: 'destructive',
   570→        })
   571→      } finally {
   572→        setVisionBusy(false)
   573→      }
   574→    },
   575→    [sending, regenerating, imageBusy, visionBusy, power, bumpConversation, toast, stopSpeaking]
   576→  )
   577→
   578→  // ---------- دوباره جواب بده ----------
   579→  const regenerate = useCallback(async () => {
   580→    if (sending || regenerating || imageBusy || visionBusy || !activeIdRef.current) return
   581→    stopSpeaking()
   582→    setRegenerating(true)
   583→    try {
   584→      const res = await fetch('/api/chat/regenerate', {
   585→        method: 'POST',
   586→        headers: { 'Content-Type': 'application/json' },
   587→        body: JSON.stringify({ conversationId: activeIdRef.current, power }),
   588→        signal: AbortSignal.timeout(150_000),
   589→      })
   590→      const data = await res.json().catch(() => null)
   591→      if (!res.ok) throw new Error(data?.error || 'جواب جدید نیومد!')
   592→
   593→      setMessages((m) => {
   594→        const copy = [...m]
   595→        for (let i = copy.length - 1; i >= 0; i--) {
   596→          if (copy[i].role === 'assistant') {
   597→            copy[i] = data.reply
   598→            break
   599→          }
   600→        }
   601→        // اگه هیچ جوابی نبود، اضافه کن
   602→        if (!copy.some((x) => x.role === 'assistant')) copy.push(data.reply)
   603→        return copy
   604→      })
   605→    } catch (err) {
   606→      toast({
   607→        title: 'اووپس!',
   608→        description: err instanceof Error ? err.message : 'یه خطا پیش اومد!',
   609→        variant: 'destructive',
   610→      })
   611→    } finally {
   612→      setRegenerating(false)
   613→    }
   614→  }, [sending, regenerating, power, toast])
   615→
   616→  // ---------- کپی ----------
   617→  const copyText = useCallback(
   618→    async (text: string) => {
   619→      try {
   620→        await navigator.clipboard.writeText(text)
   621→        toast({ title: 'کپی شد! 📋', description: 'پیام تکا توی کلیپ‌بورده.' })
   622→      } catch {
   623→        toast({ title: 'آخ!', description: 'کپی نشد!', variant: 'destructive' })
   624→      }
   625→    },
   626→    [toast]
   627→  )
   628→
   629→  // ---------- خروجی چت ----------
   630→  const exportChat = useCallback(() => {
   631→    if (messages.length === 0) {
   632→      toast({ title: 'چیزی برای خروجی نیست!', description: 'اول یه چی بپرس داداش 😄' })
   633→      return
   634→    }
   635→    const lines = [
   636→      `# چت تکا — هوش مصنوعی ساختهٔ محمد صادقی`,
   637→      ``,
   638→      ...messages.map(
   639→        (m) => `${m.role === 'user' ? '🙋 شما' : '🤖 تکا'}:\n${m.content}\n`
   640→      ),
   641→    ]
   642→    const blob = new Blob(['\uFEFF' + lines.join('\n')], {
   643→      type: 'text/plain;charset=utf-8',
   644→    })
   645→    const url = URL.createObjectURL(blob)
   646→    const a = document.createElement('a')
   647→    a.href = url
   648→    a.download = `teka-chat-${new Date().toISOString().slice(0, 10)}.txt`
   649→    a.click()
   650→    URL.revokeObjectURL(url)
   651→    toast({ title: 'خروجی گرفتی! 📥', description: 'فایل چت دانلود شد.' })
   652→  }, [messages, toast])
   653→
   654→  // ---------- تغییر تم ----------
   655→  const toggleTheme = useCallback(() => {
   656→    setTheme(theme === 'dark' ? 'light' : 'dark')
   657→  }, [theme, setTheme])
   658→
   659→  // آخرین جواب تکا (برای دکمه دوباره بگو)
   660→  const lastAssistantId = useMemo(() => {
   661→    for (let i = messages.length - 1; i >= 0; i--) {
   662→      if (messages[i].role === 'assistant') return messages[i].id
   663→    }
   664→    return null
   665→  }, [messages])
   666→
   667→  const busy = sending || regenerating || imageBusy || visionBusy
   668→
   669→  // ستاره‌های پس‌زمینه (ثابت برای جلوگیری از خطای hydration)
   670→  const stars = useMemo(
   671→    () => [
   672→      { top: '12%', left: '8%', size: 3, dur: '3.6s', delay: '0s' },
   673→      { top: '22%', left: '82%', size: 2, dur: '4.4s', delay: '0.6s' },
   674→      { top: '38%', left: '15%', size: 2, dur: '5s', delay: '1.2s' },
   675→      { top: '55%', left: '88%', size: 3, dur: '4s', delay: '0.3s' },
   676→      { top: '68%', left: '6%', size: 2, dur: '4.8s', delay: '1.8s' },
   677→      { top: '78%', left: '70%', size: 2, dur: '3.8s', delay: '0.9s' },
   678→      { top: '30%', left: '45%', size: 2, dur: '5.2s', delay: '2.1s' },
   679→      { top: '85%', left: '35%', size: 3, dur: '4.2s', delay: '1.5s' },
   680→      { top: '8%', left: '55%', size: 2, dur: '4.6s', delay: '2.4s' },
   681→      { top: '48%', left: '60%', size: 2, dur: '3.9s', delay: '0.4s' },
   682→    ],
   683→    []
   684→  )
   685→
   686→  return (
   687→    <div className="relative flex h-dvh w-full overflow-hidden">
   688→      {/* پس‌زمینهٔ aurora + ستاره‌ها */}
   689→      <div className="teka-aurora" aria-hidden="true">
   690→        <div className="teka-blob teka-blob-1" />
   691→        <div className="teka-blob teka-blob-2" />
   692→        <div className="teka-blob teka-blob-3" />
   693→        <div className="teka-blob teka-blob-4" />
   694→        {stars.map((s, i) => (
   695→          <span
   696→            key={i}
   697→            className="teka-star"
   698→            style={{
   699→              top: s.top,
   700→              left: s.left,
   701→              width: s.size,
   702→              height: s.size,
   703→              ['--dur' as string]: s.dur,
   704→              animationDelay: s.delay,
   705→            }}
   706→          />
   707→        ))}
   708→      </div>
   709→
   710→      <TekaSidebar
   711→        conversations={conversations}
   712→        activeId={activeId}
   713→        loading={loadingList}
   714→        open={sidebarOpen}
   715→        onClose={() => setSidebarOpen(false)}
   716→        onSelect={openConversation}
   717→        onNewChat={newChat}
   718→        onDelete={deleteConversation}
   719→      />
   720→
   721→      <div className="relative z-10 flex min-w-0 flex-1 flex-col">
   722→        {/* هدر */}
   723→        <header className="flex items-center gap-2 border-b border-border/60 bg-background/70 px-3 py-2.5 backdrop-blur-xl md:px-6">
   724→          <Button
   725→            variant="ghost"
   726→            size="icon"
   727→            className="md:hidden"
   728→            onClick={() => setSidebarOpen(true)}
   729→            aria-label="باز کردن تاریخچه چت‌ها"
   730→          >
   731→            <Menu className="size-5" />
   732→          </Button>
   733→          <div className="flex min-w-0 flex-1 items-center gap-2.5">
   734→            <span className="relative flex size-2.5 shrink-0" aria-hidden="true">
   735→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-primary opacity-60" />
   736→              <span className="relative inline-flex size-2.5 rounded-full bg-primary" />
   737→            </span>
   738→            <div className="min-w-0">
   739→              <p className="truncate text-sm font-bold leading-5">تکا</p>
   740→              <p className="truncate text-[11px] leading-4 text-muted-foreground">
   741→                همیشه آنلاین • حافظه داره
   742→              </p>
   743→            </div>
   744→          </div>
   745→
   746→          {/* حالت قدرتمند */}
   747→          <Button
   748→            variant={power ? 'default' : 'ghost'}
   749→            size="sm"
   750→            onClick={() => setPower((p) => !p)}
   751→            className={`gap-1.5 rounded-full text-xs font-semibold transition-all ${
   752→              power ? 'shadow-lg shadow-primary/30' : 'text-muted-foreground'
   753→            }`}
   754→            aria-pressed={power}
   755→            title="حالت قدرتمند: جواب‌های عمیق‌تر و باهوش‌تر"
   756→          >
   757→            <Zap className={`size-4 ${power ? 'fill-current' : ''}`} />
   758→            <span className="hidden sm:inline">{power ? 'قدرتمند' : 'عادی'}</span>
   759→          </Button>
   760→
   761→          {/* خواندن خودکار */}
   762→          <Button
   763→            variant={autoSpeak ? 'default' : 'ghost'}
   764→            size="icon"
   765→            onClick={() => {
   766→              setAutoSpeak((v) => {
   767→                if (v) stopSpeaking()
   768→                return !v
   769→              })
   770→            }}
   771→            className={autoSpeak ? 'shadow-lg shadow-primary/30' : 'text-muted-foreground'}
   772→            aria-pressed={autoSpeak}
   773→            aria-label="خواندن خودکار جواب‌ها"
   774→            title="خودکار بلند بخون جواب‌هارو"
   775→          >
   776→            {mounted && autoSpeak ? <Volume2 className="size-4.5" /> : <VolumeX className="size-4.5" />}
   777→          </Button>
   778→
   779→          {/* خروجی چت */}
   780→          <Button
   781→            variant="ghost"
   782→            size="icon"
   783→            onClick={exportChat}
   784→            className="text-muted-foreground"
   785→            aria-label="دانلود خروجی چت"
   786→            title="خروجی چت"
   787→          >
   788→            <Download className="size-4.5" />
   789→          </Button>
   790→
   791→          {/* تم */}
   792→          <Button
   793→            variant="ghost"
   794→            size="icon"
   795→            onClick={toggleTheme}
   796→            className="text-muted-foreground"
   797→            aria-label={mounted && theme === 'dark' ? 'تم روشن' : 'تم تیره'}
   798→            title="تغییر تم"
   799→          >
   800→            {mounted && theme === 'dark' ? <Sun className="size-4.5" /> : <Moon className="size-4.5" />}
   801→          </Button>
   802→
   803→          <span className="hidden rounded-full border border-primary/30 bg-primary/10 px-3 py-1 text-[11px] font-medium text-primary lg:inline">
   804→            ساختهٔ محمد صادقی
   805→          </span>
   806→        </header>
   807→
   808→        {/* پیام‌ها */}
   809→        <main
   810→          ref={mainRef}
   811→          onScroll={handleMainScroll}
   812→          className="teka-scroll relative flex-1 overflow-y-auto"
   813→          aria-label="گفتگو"
   814→        >
   815→          <div className="mx-auto max-w-3xl px-3 py-4 md:px-6">
   816→            {messages.length === 0 && !loadingMessages && !busy ? (
   817→              <EmptyState onSuggestion={send} power={power} />
   818→            ) : (
   819→              <div className="flex flex-col gap-4 pb-2">
   820→                {loadingMessages &&
   821→                  Array.from({ length: 3 }).map((_, i) => (
   822→                    <div
   823→                      key={i}
   824→                      className={`flex ${i % 2 ? 'justify-end' : 'justify-start'}`}
   825→                      aria-hidden="true"
   826→                    >
   827→                      <div className="h-10 w-2/3 animate-pulse rounded-2xl bg-muted/50" />
   828→                    </div>
   829→                  ))}
   830→                {!loadingMessages &&
   831→                  messages.map((m, idx) => {
   832→                    // ⭐ برای عکس‌ها پرامپت اصلی رو پیدا کن تا «دوباره بکش» کار کنه
   833→                    let redrawPrompt: string | undefined
   834→                    if (m.role === 'assistant' && m.content.startsWith('TEKA_IMG::')) {
   835→                      for (let j = idx - 1; j >= 0; j--) {
   836→                        if (messages[j].role === 'user') {
   837→                          redrawPrompt = messages[j].content
   838→                            .replace(/^🎨 درخواست تصویر:\s*/, '')
   839→                            .replace(/^TEKA_UIMG::.*\n?/, '')
   840→                            .trim()
   841→                          break
   842→                        }
   843→                      }
   844→                    }
   845→                    return (
   846→                      <MessageBubble
   847→                        key={m.id}
   848→                        message={m}
   849→                        isLastAssistant={m.id === lastAssistantId && !busy}
   850→                        onCopy={copyText}
   851→                        onSpeak={speak}
   852→                        onRegenerate={regenerate}
   853→                        onRedraw={
   854→                          redrawPrompt && !busy
   855→                            ? () => sendImage(redrawPrompt)
   856→                            : undefined
   857→                        }
   858→                        speakingId={speakingId}
   859→                        speakingLoadingId={speakingLoadingId}
   860→                      />
   861→                    )
   862→                  })}
   863→                {(imageBusy || awaitingArt) && <PaintingPlaceholder />}
   864→                {!imageBusy && !awaitingArt && busy && (
   865→                  <TypingIndicator power={power} analyzing={visionBusy} />
   866→                )}
   867→              </div>
   868→            )}
   869→            <div ref={bottomRef} />
   870→          </div>
   871→
   872→          {/* دکمه اسکرول به پایین */}
   873→          <AnimatePresence>
   874→            {showScrollBtn && (
   875→              <motion.div
   876→                initial={{ opacity: 0, y: 10 }}
   877→                animate={{ opacity: 1, y: 0 }}
   878→                exit={{ opacity: 0, y: 10 }}
   879→                className="sticky bottom-4 flex justify-center"
   880→              >
   881→                <Button
   882→                  size="icon"
   883→                  onClick={() => bottomRef.current?.scrollIntoView({ behavior: 'smooth' })}
   884→                  className="size-10 rounded-full border bg-card shadow-xl"
   885→                  aria-label="برو پایین"
   886→                >
   887→                  <ArrowDown className="size-4" />
   888→                </Button>
   889→              </motion.div>
   890→            )}
   891→          </AnimatePresence>
   892→        </main>
   893→
   894→        {/* اینپوت (فوتر چسبیده به پایین) */}
   895→        <ChatInput
   896→          value={input}
   897→          onChange={setInput}
   898→          onSend={send}
   899→          onSendImage={sendImage}
   900→          onSendVision={sendVision}
   901→          onSendFile={sendFile}
   902→          sending={busy}
   903→          analyzing={visionBusy}
   904→          power={power}
   905→        />
   906→      </div>
   907→    </div>
   908→  )
   909→}
